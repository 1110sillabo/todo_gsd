**Simpler approach (use this):** Connect task_edit_requested per-row inside GroupSection.add_task(). In task_list_view.gd, store `_editing_task_id` on open; on save, find and refresh the row:

```gdscript
var _editing_task_id: String = ""

func _on_task_edit_requested(task: TaskResource) -> void:
	_editing_task_id = task.task_id
	_edit_dialog.show_for_task(task)

func _on_task_saved(task: TaskResource) -> void:
	PersistenceManager.save_task(task)
	_refresh_or_move_row(task)

func _refresh_or_move_row(task: TaskResource) -> void:
	var now: int = int(Time.get_unix_time_from_system())
	var new_category: String = categorize(task, now)
	# Find row in each group and recategorize if needed
	for group in [group_expired, group_todo, group_completed]:
		var row: TaskRow = group.find_row(task.task_id)
		if row != null:
			var old_category: String = _group_name(group)
			if old_category == new_category:
				row.set_task(task)  # refresh in-place (per D-05)
			else:
				group.remove_task(task)
				_add_to_group(new_category, task)
			return

func _group_name(group: GroupSection) -> String:
	if group == group_expired:
		return "expired"
	elif group == group_completed:
		return "completed"
	return "todo"

func _add_to_group(category: String, task: TaskResource) -> void:
	match category:
		"expired":
			group_expired.add_task(task)
		"completed":
			group_completed.add_task(task)
		_:
			group_todo.add_task(task)
```

| TaskEditDialog → TaskResource fields | User-edited strings overwrite task title/description/deadline |
| TaskResource → PersistenceManager | Updated resource written to disk via save_task() |

| Threat ID | Category | Component | Disposition | Mitigation Plan |

| T-03-04 | Tampering | task_edit_requested: wrong row reference | mitigate | Use task_id (not row pointer) to find row on save; avoids stale reference bugs |
| T-03-05 | Denial | Null task reference in _on_task_edit_requested | mitigate | Guard: `if task == null: return` at top of _on_task_edit_requested |
